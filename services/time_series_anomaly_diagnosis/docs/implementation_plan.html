def file_sha256(path: Path) -> str:
    return sha256(path.read_bytes()).hexdigest()

def build_code_map() -> dict[str, object]:
    nodes = []
    for relationship in RELATIONSHIPS:
        path = SERVICE_ROOT / relationship["path"]
        nodes.append({**relationship, "sha256": file_sha256(path)})
    return {
        "schema": "time_series_anomaly_diagnosis_code_map_v1",
        "nodes": nodes,
        "edges": [],
        "unknown_relationships": [],

def render_html(payload: dict[str, object]) -> str:
    rows = "".join(
        f"<tr><td>{escape(node['id'])}</td><td>{escape(node['path'])}</td>"
        f"<td>{escape(node['caller'])}</td><td>{escape(node['effect'])}</td></tr>"
        for node in payload["nodes"]
    )
    return (
        f"<!doctype html><html lang='ko'><meta charset='utf-8'>"
        f"<title>서비스 코드맵 v1</title><table>{rows}</table></html>"
    )

def canonical_json(payload: object) -> bytes:
    return (json.dumps(payload, ensure_ascii=False, sort_keys=True, indent=2) + "\n").encode()

def write_code_map() -> None:
    payload = build_code_map()
    json_bytes = canonical_json(payload)
    JSON_PATH.write_bytes(json_bytes)
    HTML_PATH.write_text(render_html(payload), encoding="utf-8")
    LOCK_PATH.write_bytes(canonical_json({"map_sha256": sha256(json_bytes).hexdigest()}))

    payload = build_code_map()
    expected_json = canonical_json(payload)

def open_source_database(path: Path = SOURCE_DATABASE) -> sqlite3.Connection:
    connection = sqlite3.connect(f"{path.resolve().as_uri()}?mode=ro", uri=True, check_same_thread=False)
    connection.row_factory = sqlite3.Row
    connection.execute("PRAGMA query_only=ON")

def get_dataset_catalog(connection: sqlite3.Connection, resources: ModelResources) -> list[dict[str, object]]:
    entries = []
    rows = connection.execute(
        "SELECT name, train_total_count, test_total_count, series_length FROM datasets ORDER BY name"
    )
    for row in rows:
        dataset_name = row["name"]
        method_name = resources.route_by_dataset.get(dataset_name)
        if method_name is None:
            availability = "missing_verified_route"
        elif dataset_name in resources.recovery_datasets:
            availability = "recovery_numba"
        elif row["train_total_count"] < 5:
            availability = "insufficient_training_data"
        else:
            availability = "standard"
        entries.append({
            "dataset_name": dataset_name,
            "method_name": method_name,
            "train_count": int(row["train_total_count"]),
            "test_count": int(row["test_total_count"]),
            "series_length": int(row["series_length"]),
            "availability": availability,
            "analysis_available": availability in {"standard", "recovery_numba"},
        })
    return entries

def standardize_equal_length_rows(rows: list[np.ndarray]) -> np.ndarray:
    if not rows or len({row.size for row in rows}) != 1:
        raise ValueError("standard preparation requires non-empty equal-length rows")
    values = np.asarray(rows, dtype=np.float64)
    scales = values.std(axis=1, keepdims=True)
    if not np.isfinite(values).all() or np.any(scales == 0.0):
        raise ValueError("standard preparation requires finite non-constant rows")
    return (values - values.mean(axis=1, keepdims=True)) / scales

def load_dataset_values(connection: sqlite3.Connection, dataset_name: str, preparation_mode: str) -> PreparedDatasetValues:
    if preparation_mode == "recovery_numba":
        prepared = load_recovery_values(connection, dataset_name)
        return PreparedDatasetValues(dataset_name, prepared.train_rows, prepared.test_rows, prepared.test_indices, preparation_mode)
    if preparation_mode != "standard":
        raise ValueError("dataset is not available for analysis")
    train, test, test_indices = [], [], []
    expected_index = {"TRAIN": 0, "TEST": 0}
    for split, instance_index, blob in connection.execute(VALUE_SQL, (dataset_name,)):
        if split not in expected_index or instance_index != expected_index[split]:
            raise ValueError("source value rows are not sequential")
        expected_index[split] += 1
        if type(blob) is not bytes or not blob or len(blob) % 4:
            raise ValueError("source value blob is invalid")
        values = np.frombuffer(blob, dtype="<f4")
        (train if split == "TRAIN" else test).append(values)
        if split == "TEST":
            test_indices.append(instance_index)
    if len({row.size for row in train + test}) != 1:
        raise ValueError("standard TRAIN and TEST lengths must match")
    return PreparedDatasetValues(
        dataset_name,
        standardize_equal_length_rows(train),
        standardize_equal_length_rows(test),
        tuple(test_indices),
        preparation_mode,
    )

function chartPoint(canvas, index, length, value, minimum, maximum) {
  const x = index * canvas.width / Math.max(length - 1, 1);
  const y = canvas.height - (value - minimum) * canvas.height / Math.max(maximum - minimum, Number.EPSILON);
  return [x, y];
}

function drawLine(context, values, minimum, maximum, color) {
  context.beginPath();
  values.forEach((value, index) => {
    const [x, y] = chartPoint(context.canvas, index, values.length, value, minimum, maximum);
    if (index === 0) context.moveTo(x, y); else context.lineTo(x, y);
  });
  context.strokeStyle = color;
  context.lineWidth = 2;
  context.stroke();
}

function drawReferenceBand(context, lower, upper, minimum, maximum) {
  context.beginPath();
  upper.forEach((value, index) => {
    const [x, y] = chartPoint(context.canvas, index, upper.length, value, minimum, maximum);
    if (index === 0) context.moveTo(x, y); else context.lineTo(x, y);
  });
  for (let index = lower.length - 1; index >= 0; index -= 1) {
    const [x, y] = chartPoint(context.canvas, index, lower.length, lower[index], minimum, maximum);
    context.lineTo(x, y);
  }
  context.closePath();
  context.fillStyle = "rgba(83, 116, 201, 0.18)";
  context.fill();
}
