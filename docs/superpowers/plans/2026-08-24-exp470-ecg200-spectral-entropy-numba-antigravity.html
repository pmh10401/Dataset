Exp470 ECG200 spectral entropy Numba 구현 계획

For agentic workers: 구현 시 superpowers:subagent-driven-development(권장) 또는 superpowers:executing-plans를 반드시 사용한다. 각 단계는 아래 HTML checkbox로 추적한다.

목표: ECG200 TRAIN-only에서 spectral entropy가 C0의 잔여 정상–period 순서 오류를 안전하게 고치는지 Numba 단일 계산 경로로 검증한다.

구조: Exp468의 입력·보간·합성 변화·C0 authority를 그대로 가져오고, Exp470은 spectral entropy, C11 결합, pair repair 판정만 새로 구현한다. 별도 NumPy 점수 backend는 만들지 않으며 작은 Python math 기준 검사와 재현성 검사를 통과해야 결과 실행 경로가 열린다.

기술: Python 3, Numba, llvmlite, 기존 ndarray 기반 Exp468 authority, unittest, canonical JSON/JSONL, self-contained HTML.

설계서: Exp470 승인 설계서

전역 제약

  • 새 파일 이름에는 모두 antigravity가 포함되어야 한다.
  • ECG200 TEST, 실제 FN/FP/F1, routing, champion, pending, promotion을 열거나 변경하지 않는다.
  • 새 점수 후보는 E11 spectral entropy 하나뿐이다.
  • 새 점수의 공식 backend는 Numba 하나다. 별도 NumPy scorer와 자동 fallback을 만들지 않는다.
  • 길이 28 direct DFT, float64, fastmath=False, cache=False, epsilon 1e-12, 자연로그를 고정한다.
  • Python math 기준 대비 raw 최대 오차는 1e-12 이하여야 하며 rank·confirmation·fixed/broken은 정확히 같아야 한다.
  • MLX는 전체 실행 전에 적격성만 평가한다. Numba 공식 계산을 대신하지 않는다.
  • private 실행 1·2와 official 실행은 이 계획의 구현 Task에 포함하지 않으며 별도 사용자 승인이 필요하다.
  • 각 Task는 실패 테스트 확인 → 최소 구현 → 통과 확인 → 독립 검토 → 커밋 순서로 끝낸다.

def entropy_window_python_antigravity(window: Sequence[float]) -> float:
    mean = math.fsum(window) / 28.0
    weighted = [(window[n] - mean) * (0.5 - 0.5 * math.cos(2.0 * math.pi * n / 27.0)) for n in range(28)]
    power = []
    for k in range(1, 15):
        real = math.fsum(weighted[n] * math.cos(2.0 * math.pi * k * n / 28.0) for n in range(28))
        imag = -math.fsum(weighted[n] * math.sin(2.0 * math.pi * k * n / 28.0) for n in range(28))
        power.append(real * real + 1e-12)
    total = math.fsum(power)
    probabilities = [value / total for value in power]
    return -math.fsum(p * math.log(p) for p in probabilities) / math.log(14.0)

def entropy_row_python_antigravity(row: Sequence[float]) -> float:
    starts = (0, 14, 28, 42, 56)
    values = [entropy_window_python_antigravity(row[56 + start:56 + start + 28]) for start in starts]
    return float(statistics.median(values))

def score_entropy_python_antigravity(fit_rows, query_rows) -> PythonReferenceScoresAntigravity:
    fit_h = [entropy_row_python_antigravity(tuple(row)) for row in fit_rows]
    query_h = [entropy_row_python_antigravity(tuple(row)) for row in query_rows]
    fit_raw = []
    for excluded, value in enumerate(fit_h):
        others = [candidate for index, candidate in enumerate(fit_h) if index != excluded]
        fit_raw.append(abs(value - statistics.median(others)))
    center = statistics.median(fit_h)
    query_raw = [abs(value - center) for value in query_h]
    maximum = max(fit_raw)
    ranks = tuple((1 + sum(value <= raw for value in fit_raw)) / (len(fit_raw) + 1) for raw in query_raw)
    confirmed = tuple(raw > 0.0 and raw > maximum for raw in query_raw)
    return PythonReferenceScoresAntigravity(tuple(fit_raw), tuple(query_raw), ranks, confirmed, maximum)

@numba.njit(cache=False, fastmath=False)
def strict_confirmation_numba_antigravity(query_raw, maximum):
    output = np.empty(query_raw.size, dtype=np.bool_)
    for index in range(query_raw.size):
        output[index] = query_raw[index] > 0.0 and query_raw[index] > maximum
    return output

@numba.njit(cache=False, fastmath=False)
def entropy_rows_numba_antigravity(rows: np.ndarray) -> np.ndarray:
    output = np.empty(rows.shape[0], dtype=np.float64)
    starts = (0, 14, 28, 42, 56)
    weighted = np.empty(28, dtype=np.float64)
    frame_entropy = np.empty(5, dtype=np.float64)
    power = np.empty(14, dtype=np.float64)
    for row_index in range(rows.shape[0]):
        for frame_index in range(5):
            base = 56 + starts[frame_index]
            mean = 0.0
            for n in range(28):
                mean += rows[row_index, base + n]
            mean /= 28.0
            for n in range(28):
                hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * n / 27.0)
                weighted[n] = (rows[row_index, base + n] - mean) * hann
            total = 0.0
            for bin_index in range(14):
                k = bin_index + 1
                real = 0.0
                imag = 0.0
                for n in range(28):
                    angle = 2.0 * np.pi * k * n / 28.0
                    real += weighted[n] * np.cos(angle)
                    imag -= weighted[n] * np.sin(angle)
                value = real * real + imag * imag + 1e-12
                power[bin_index] = value
                total += value
            entropy = 0.0
            for bin_index in range(14):
                probability = power[bin_index] / total
                entropy -= probability * np.log(probability)
            frame_entropy[frame_index] = entropy / np.log(14.0)
        output[row_index] = np.median(frame_entropy)
    return output

@numba.njit(cache=False, fastmath=False)
def _score_entropy_arrays_numba_antigravity(fit_rows, query_rows):
    fit_h = entropy_rows_numba_antigravity(fit_rows)
    query_h = entropy_rows_numba_antigravity(query_rows)
    fit_raw = np.empty(fit_h.size, dtype=np.float64)
    loo = np.empty(fit_h.size - 1, dtype=np.float64)
    for excluded in range(fit_h.size):
        position = 0
        for index in range(fit_h.size):
            if index != excluded:
                loo[position] = fit_h[index]
                position += 1
        fit_raw[excluded] = abs(fit_h[excluded] - np.median(loo))
    center = np.median(fit_h)
    maximum = np.max(fit_raw)
    query_raw = np.empty(query_h.size, dtype=np.float64)
    ranks = np.empty(query_h.size, dtype=np.float64)
    confirmed = np.empty(query_h.size, dtype=np.bool_)
    for query_index in range(query_h.size):
        raw = abs(query_h[query_index] - center)
        query_raw[query_index] = raw
        less_equal = 0
        for fit_index in range(fit_raw.size):
            if fit_raw[fit_index] <= raw:
                less_equal += 1
        ranks[query_index] = (1.0 + less_equal) / (fit_raw.size + 1.0)
        confirmed[query_index] = raw > 0.0 and raw > maximum
    return fit_raw, query_raw, ranks, confirmed, maximum

def score_spectral_entropy_numba_antigravity(fit_rows, query_rows) -> EntropyScoresAntigravity:
    fit = _validated_rows_antigravity(fit_rows, "fit_rows")
    query = _validated_rows_antigravity(query_rows, "query_rows")
    arrays = _score_entropy_arrays_numba_antigravity(fit, query)
    return EntropyScoresAntigravity(*arrays, backend="numba")

    fixed = 0
    broken = 0
    for changed_index in range(c0_changed.size):
        for normal_index in range(c0_normal.size):
            before = _credit_antigravity(c0_normal[normal_index], c0_changed[changed_index])
            after = _credit_antigravity(c11_normal[normal_index], c11_changed[changed_index])
            if after > before:
                fixed += 1
            elif after < before:
                broken += 1
    return fixed, broken
