def mann_whitney_auc_antigravity(normal, anomaly):
    normal = exact_finite_vector_antigravity(normal)
    anomaly = exact_finite_vector_antigravity(anomaly)
    wins = np.count_nonzero(anomaly[:, None] > normal[None, :])
    ties = np.count_nonzero(anomaly[:, None] == normal[None, :])
    return float((wins + 0.5 * ties) / (len(normal) * len(anomaly)))
